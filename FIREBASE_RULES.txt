FIRESTORE RULES
===============
Use these rules for the Smart School demo project. They require Firebase Anonymous Authentication.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /smartSchoolMedia/{docId} {
      allow read, write: if request.auth != null;
    }
    match /smartSchoolArduino/{docId} {
      allow read, write: if request.auth != null;
    }
  }
}

STORAGE RULES
=============
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /smart-school-media/{allPaths=**} {
      allow read, write: if request.auth != null;
    }
  }
}
